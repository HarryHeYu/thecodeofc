#词嵌入与位置编码
import torch
import torch.nn as nn
import math

class PositionalEncoding(nn.Module):
    def __init__(self, d_model: int, max_len: int = 512):
        super().__init__()
        self.d_model = d_model
        # 生成位置编码矩阵（不参与梯度更新）
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).
        unsqueeze(1)  # [max_len, 1]
        div_term = torch.exp(torch.arange(0, d_model, 2).
        float() * (-math.log(10000.0) / d_model))
        
        pe[:, 0::2] = torch.sin(position * div_term)  # 偶数维度用sin
        pe[:, 1::2] = torch.cos(position * div_term)  # 奇数维度用cos
        self.register_buffer('pe', pe)  # 注册为缓冲区（固定参数）

    def forward(self, x: torch.Tensor):
        # x: [batch_size, seq_len, d_model]
        x = x + self.pe[:x.size(1)]  # 相加位置编码（自动广播）
        return x

class WordEmbedding(nn.Module):
    def __init__(self, vocab_size: int, d_model: int):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, d_model)  # 词嵌入层
        self.d_model = d_model

    def forward(self, x: torch.Tensor):
        # x: [batch_size, seq_len] 词元索引序列
        return self.embedding(x) * math.sqrt(self.d_model) 
         # 缩放初始化（提高稳定性）

# 自注意力机制（含多头注意力）
class ScaledDotProductAttention(nn.Module):
    def __init__(self):
        super().__init__()
    
    def forward(self, Q: torch.Tensor, K: torch.Tensor, 
    V: torch.Tensor, mask: torch.Tensor = None):
        # Q, K, V: [batch_size, n_heads, seq_len, d_k]
        attn_scores = torch.matmul(Q, K.transpose(-2, -1)) / 
        math.sqrt(Q.size(-1))  # 缩放点积 [B, h, L, L]
        
        if mask is not None:  # 掩码（用于解码器的未来位置遮挡）
            attn_scores = attn_scores.masked_fill(mask == 0, -1e9)
        
        attn_weights = nn.functional.softmax(attn_scores, dim=-1)  
        # 注意力权重
        output = torch.matmul(attn_weights, V)  
        # 加权聚合Value [B, h, L, d_v]
        return output, attn_weights

class MultiHeadAttention(nn.Module):
    def __init__(self, d_model: int, n_heads: int = 8):
        super().__init__()
        self.n_heads = n_heads
        self.d_k = d_model // n_heads  # 每个头的维度       
        # 线性变换层：生成Q, K, V
        self.W_Q = nn.Linear(d_model, d_model)
        self.W_K = nn.Linear(d_model, d_model)
        self.W_V = nn.Linear(d_model, d_model)
        self.W_O = nn.Linear(d_model, d_model)  # 输出线性层

    def forward(self, x: torch.Tensor, mask: torch.Tensor = None):
        # x: [batch_size, seq_len, d_model]
        batch_size = x.size(0)
        seq_len = x.size(1)
        
        Q = self.W_Q(x).view(batch_size, seq_len, self.n_heads, self.d_k).transpose(1, 2)
        K = self.W_K(x).view(batch_size, seq_len, self.n_heads, self.d_k).transpose(1, 2)
        V = self.W_V(x).view(batch_size, seq_len, self.n_heads, self.d_k).transpose(1, 2)
        
        # 缩放点积注意力
        output, attn_weights = ScaledDotProductAttention()(Q, K, V, mask)  # [B, h, L, d_k]
        output = output.transpose(1, 2).contiguous().view(batch_size, seq_len, -1)
        output = self.W_O(output)
        return output, attn_weights

#前馈神经网络（FFN）
class FeedForwardNetwork(nn.Module):
    def __init__(self, d_model: int, d_ff: int = 2048, activation: str = 'gelu'):
        super().__init__()
        self.fc1 = nn.Linear(d_model, d_ff)
        self.fc2 = nn.Linear(d_ff, d_model)
        self.activation = activation.lower()
    
    def forward(self, x: torch.Tensor):
        if self.activation == 'gelu':
            x = nn.functional.gelu(self.fc1(x))  # GELU激活函数
        elif self.activation == 'relu':
            x = nn.functional.relu(self.fc1(x))
        return self.fc2(x)

#编码器层（含残差连接与层归一化）
class EncoderLayer(nn.Module):
    def __init__(self, d_model: int, n_heads: int, d_ff: int):
        super().__init__()
        self.self_attn = MultiHeadAttention(d_model, n_heads)
        self.ffn = FeedForwardNetwork(d_model, d_ff)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(0.1)
    
    def forward(self, x: torch.Tensor, mask: torch.Tensor = None):
        # 自注意力子层
        attn_output, _ = self.self_attn(x, mask)
        x = self.norm1(x + self.dropout(attn_output))  # 残差连接+层归一化
        
        # 前馈神经网络子层
        ffn_output = self.ffn(x)
        x = self.norm2(x + self.dropout(ffn_output))  # 残差连接+层归一化
        return x
